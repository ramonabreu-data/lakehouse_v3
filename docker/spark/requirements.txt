# --- travas de compatibilidade -------------------------------------------
# A imagem base vem com protobuf 7.x, mas o TensorFlow 2.17 exige <5. Sem esta
# trava o TF importa cuspindo "MessageFactory object has no attribute
# GetPrototype". A faixa abaixo satisfaz TF, google-api-core e proto-plus.
protobuf>=4.25,<5
# numpy 2.x quebra o tensorflow 2.17 e o torch 2.4 desta imagem.
numpy<2
